        self.reset()

    def reset(self):
        self.step_no = 0
        self.max_step = 1000
        self.last_action = None
        self.last_min_monster_dist = 0.5
        self.last_treasure_dist = None

    def feature_process(self, env_obs, last_action):
        """完整的特征提取"""
        observation = env_obs["observation"]
        frame_state = observation["frame_state"]
        env_info = observation["env_info"]
        map_info = observation["map_info"]
        legal_act_raw = observation["legal_action"]

        self.step_no = observation["step_no"]
        self.max_step = env_info.get("max_step", 1000)

        # 更新地图中心坐标（map_info以英雄为中心）
        if map_info is not None:
            self.hero_center_x = len(map_info) // 2
            self.hero_center_z = len(map_info) // 2

        # ========== 1. 英雄自身特征 (6D) ==========
        hero = frame_state["heroes"]
        hero_pos = hero["pos"]

        hero_feat = np.array([
            hero_pos["x"] / MAP_SIZE,  # 位置归一化
            hero_pos["z"] / MAP_SIZE,
            hero["treasure_score"] / 1000.0,  # 分数归一化（假设最大1000）
            hero["step_score"] / (self.max_step * 1.5),  # 步数分归一化
            1.0 if hero["flash_cooldown"] == 0 else 0.0,  # 闪现是否可用
            hero["buff_remaining_time"] / MAX_BUFF_DURATION,  # buff剩余时间
        ], dtype=np.float32)

        # ========== 2. 怪物特征 (6D × 2) ==========
        monsters = frame_state.get("monsters", [])

        monster_feats = []
        for i in range(2):
            if i < len(monsters):
                m = monsters[i]
                is_in_view = float(m.get("is_in_view", 0))
                speed = m.get("speed", 1) / MAX_MONSTER_SPEED
                m_pos = m["pos"]

                # 复合真实距离：BFS优先，离散桶兜底
                real_dist_norm = self._compute_real_distance(hero_pos, m_pos, map_info, m)
                direction = m.get("hero_relative_direction", 0) / 8.0  # 0-8 -> 0-1

                monster_feats.append(np.array([
                    real_dist_norm,   # 复合真实距离(BFS可达?BFS距离:离散桶)
                    direction,       # 相对方向(0-8归一化)
                    speed,           # 速度
                    m_pos["x"] / MAP_SIZE,  # 位置
                    m_pos["z"] / MAP_SIZE,
                    is_in_view,      # 是否在视野内
                ], dtype=np.float32))
            else:
                monster_feats.append(np.zeros(6, dtype=np.float32))

        # ========== 3. 宝箱特征 (8D) ==========
        treasures = frame_state.get("treasures", [])
        nearest_treasure = self._find_nearest_treasure(treasures, hero_pos)

        treasure_feat = np.zeros(8, dtype=np.float32)
        if nearest_treasure:
            dist_bucket = nearest_treasure.get("hero_l2_distance", 5) / 5.0
            direction = nearest_treasure.get("hero_relative_direction", 0) / 8.0
            treasure_feat = np.array([
                dist_bucket,
                direction,
                nearest_treasure["pos"]["x"] / MAP_SIZE,
                nearest_treasure["pos"]["z"] / MAP_SIZE,
                1.0,  # 宝箱是否有效
                0.0,  # 危险度（后续计算）
                0.0,  # 是否值得追（后续计算）
                0.0,  # 相对距离
            ], dtype=np.float32)

        # ========== 4. Buff特征 (4D) ==========
        buffs = frame_state.get("organs", [])
        nearest_buff = self._find_nearest_buff(buffs, hero_pos)

        buff_feat = np.zeros(4, dtype=np.float32)
        if nearest_buff:
            buff_feat = np.array([
                1.0 if hero["buff_remaining_time"] > 0 else 0.0,  # 是否拥有buff
                hero["buff_remaining_time"] / MAX_BUFF_DURATION,
                nearest_buff.get("hero_relative_direction", 0) / 8.0,
                nearest_buff.get("hero_l2_distance", 5) / 5.0,
            ], dtype=np.float32)

        # ========== 5. 技能特征 (3D) ==========
        # 怪物加速倒计时
        monster_speedup_config = env_info.get("monster_speed", 500)
        steps_until_speedup = max(0, monster_speedup_config - self.step_no) / monster_speedup_config

        skill_feat = np.array([
            hero["flash_cooldown"] / MAX_FLASH_CD,  # 闪现冷却
            1.0 if hero["flash_cooldown"] == 0 else 0.0,  # 闪现是否可用
            steps_until_speedup,  # 距离怪物加速还有多久
        ], dtype=np.float32)

        # ========== 6. 地形特征 (16D) ==========
        terrain_feat = self._compute_terrain_features(map_info, hero_pos)

        # ========== 7. 进度特征 (4D) ==========
        progress_feat = np.array([
            self.step_no / self.max_step,  # 当前进度
            1.0 if self.step_no > monster_speedup_config else 0.0,  # 是否已加速
            self._compute_danger_level(monster_feats),  # 危险等级
            self.stuck_counter / 10.0,  # 卡住程度
        ], dtype=np.float32)

        # ========== 8. 合法动作掩码 (16D) ==========
        legal_action = self._process_legal_action(legal_act_raw)

        # ========== 拼接所有特征 ==========
        feature = np.concatenate([
            hero_feat,
            monster_feats[0],
            monster_feats[1],
            treasure_feat,
            buff_feat,
            skill_feat,
            terrain_feat,
            progress_feat,
            legal_action,
        ])

        # ========== 计算奖励 ==========
        reward = self._compute_reward(
            env_obs, hero, monster_feats, treasure_feat,
            nearest_treasure, buffs, last_action, self.last_action,
            terrain_feat
        )

        # 更新历史
        self.last_action = action
        self.last_hero_pos = hero_pos.copy()
        self.history_positions.append((hero_pos["x"], hero_pos["z"]))
        if len(self.history_positions) > 20:
            self.history_positions.pop(0)

        return feature, legal_action, reward

```python
    def _compute_reward(self, env_obs, hero, monster_feats, treasure_feat,
                       nearest_treasure, buffs, current_action, last_action,
                       terrain_feat):
        """计算综合奖励"""
        rewards = []

        # ===== 1. 生存奖励（稠密）=====
        survive_reward = 0.01
        rewards.append(survive_reward)

        # ===== 2. 步数分奖励（稠密）=====
        current_step_score = hero["step_score"]
        if hasattr(self, 'last_step_score'):
            step_score_diff = current_step_score - self.last_step_score
            if step_score_diff > 0:
                rewards.append(step_score_diff * 0.1)  # 步数分上涨时给奖励
        self.last_step_score = current_step_score

        # ===== 3. 宝箱分奖励（稀疏）=====
        current_treasure_score = hero["treasure_score"]
        if hasattr(self, 'last_treasure_score'):
            treasure_diff = current_treasure_score - self.last_treasure_score
            if treasure_diff > 0:
                rewards.append(treasure_diff * 1.0)  # 宝箱分上涨时给奖励
        self.last_treasure_score = current_treasure_score

        # ===== 4. Buff奖励（稀疏）=====
        current_buff_count = len([b for b in buffs if b["status"] == 1])
        if current_buff_count > self.last_buff_count:
            rewards.append(0.5)  # 获得新buff时给奖励
        self.last_buff_count = current_buff_count

        # ===== 5. 怪物距离 shaping（稠密）=====
        cur_min_dist = 1.0
        for m_feat in monster_feats:
            if m_feat[5] > 0:  # is_in_view
                cur_min_dist = min(cur_min_dist, m_feat[0])  # dist_bucket

        if hasattr(self, 'last_min_monster_dist'):
            dist_delta = self.last_min_monster_dist - cur_min_dist  # 远离加分
            dist_reward = 0.1 * dist_delta
            rewards.append(dist_reward)

            # 加速后增加权重
            if self.step_no > self.env_info.get("monster_speed", 500):
                rewards.append(dist_reward * 2.0)  # 后期权重加倍
        self.last_min_monster_dist = cur_min_dist

        # ===== 6. 宝箱接近奖励（稠密）=====
        if nearest_treasure and treasure_feat[4] > 0:  # 宝箱有效
            current_treasure_dist = treasure_feat[0]  # dist_bucket
            if hasattr(self, 'last_treasure_dist') and self.last_treasure_dist is not None:
                treasure_delta = self.last_treasure_dist - current_treasure_dist
                # 只有在接近且不太危险时才给奖励
                if treasure_delta > 0 and cur_min_dist > 0.3:
                    rewards.append(treasure_delta * 0.5)
            self.last_treasure_dist = current_treasure_dist

        # ===== 7. 临近加速缓冲奖励（稠密）=====
        monster_speedup_config = self.env_info.get("monster_speed", 500)
        steps_until_speedup = monster_speedup_config - self.step_no
        if 0 < steps_until_speedup < 100:  # 快加速时
            buffer_reward = 0.05 * (1.0 - steps_until_speedup / 100.0)
            rewards.append(buffer_reward)

        # ===== 8. 后期生存奖励（稠密）=====
        if self.step_no > monster_speedup_config:
            # 怪物加速后，增加拉距离的奖励权重
            if hasattr(self, 'last_min_monster_dist'):
                if cur_min_dist > self.last_min_monster_dist:
                    rewards.append(0.2)  # 远离怪物

        # ===== 9. 走廊惩罚（稠密）=====
        # 走廊检测已移到 _compute_terrain_features，在奖励中统一计算

        # ===== 10. 包夹惩罚（稠密）=====
        if len(monster_feats) >= 2 and monster_feats[0][5] > 0 and monster_feats[1][5] > 0:
            # 两只怪物都在视野内，检查夹角
            m1_dir = monster_feats[0][1]  # direction
            m2_dir = monster_feats[1][1]
            angle_diff = abs(m1_dir - m2_dir)
            if angle_diff > 0.7 and angle_diff < 0.9:  # 大约90-120度
                rewards.append(-0.1)  # 被夹击惩罚

        # ===== 11. 地形奖励/惩罚（稠密）=====
        # 包含：死角惩罚、走廊惩罚、开阔区域奖励
        rewards.append(self._compute_terrain_reward(terrain_feat))

        # ===== 12. 危险惩罚（稠密）=====
        if cur_min_dist < 0.2:  # 怪物很近
            danger_penalty = -0.3 * (1.0 - cur_min_dist / 0.2)
            rewards.append(danger_penalty)

        # ===== 13. 撞墙/无效移动惩罚（稀疏）=====
        hero_pos = hero["pos"]
        if self.last_hero_pos:
            dx = hero_pos["x"] - self.last_hero_pos["x"]
            dz = hero_pos["z"] - self.last_hero_pos["z"]
            if abs(dx) + abs(dz) < 0.1:  # 没有有效移动
                rewards.append(-0.1)

        # ===== 14. 重复探索惩罚（稠密）=====
        if self._is_exploring_repeatedly():
            rewards.append(-0.05)

        # ===== 15. 闪现脱险奖励（稀疏）=====
        if current_action and current_action >= 8:  # 使用闪现
            if self.last_min_monster_dist > cur_min_dist + 0.1:  # 脱险了
                rewards.append(1.0)
            else:
                rewards.append(-0.5)  # 滥用惩罚

        # ===== 16. 第二只怪压力惩罚（稠密）=====
        if len(monster_feats) >= 2 and monster_feats[1][5] > 0:  # 第二只怪在视野内
            m2_dist = monster_feats[1][0]
            if m2_dist < 0.3:
                rewards.append(-0.2)

        return [sum(rewards)]
```

    def _compute_terrain_features(self, map_info, hero_pos):
        """
        计算地形特征
        返回: (11D) = 8方向通路深度(8) + 死角(1) + 走廊(1) + 开阔(1)

        方向索引：
            2   3
          4  hero  0
            5   6   7
        """
        if map_info is None:
            return np.zeros(11, dtype=np.float32)

        # 8方向通路深度
        direction_depths = []
        for direction in range(8):
            depth = self._trace_path(hero_pos, direction, map_info)
            direction_depths.append(depth / 10.0)  # 归一化，假设最大10步

        # 死角检测：只有1条逃生路
        escape_count = sum(1 for d in direction_depths if d >= 0.3)
        is_dead = 1.0 if escape_count <= 1 else 0.0

        # 走廊检测：两个相对方向能走，但侧向不能走
        vertical = min(direction_depths[2], direction_depths[6])  # 上+下
        horizontal = min(direction_depths[4], direction_depths[0])  # 左+右
        is_corridor = 1.0 if vertical >= 0.5 and horizontal <= 0.1 else 0.0

        # 开阔区域检测：四面八方都能走比较远
        total_depth = sum(direction_depths)
        is_open = 1.0 if total_depth >= 3.0 else 0.0

        return np.array(direction_depths + [is_dead, is_corridor, is_open], dtype=np.float32)

    def _trace_path(self, hero_pos, direction, map_info):
        """
        从起点沿指定方向发射射线，返回能走的最大深度

        direction: 0-7 (右,右上,上,左上,左,左下,下,右下)
        """
        # 方向向量
        direction_vectors = [
            (1, 0),   # 右
            (1, -1),  # 右上
            (0, -1),  # 上
            (-1, -1), # 左上
            (-1, 0),  # 左
            (-1, 1),  # 左下
            (0, 1),   # 下
            (1, 1),   # 右下
        ]

        dx, dz = direction_vectors[direction]
        x, z = hero_pos["x"], hero_pos["z"]

        depth = 0
        max_depth = 20  # 最多检查20步

        while depth < max_depth:
            next_x = x + dx * (depth + 1)
            next_z = z + dz * (depth + 1)

            # 检查是否越界或撞墙
            if not self._is_passable(next_x, next_z, map_info):
                break

            depth += 1

        return float(depth)

    def _is_passable(self, x, z, map_info):
        """检查指定坐标是否可通行（基于局部地图）"""
        # map_info 是以英雄为中心的局部地图
        # 需要将全局坐标转换为局部坐标
        center = len(map_info) // 2
        local_x = int(x - self.hero_center_x + center)
        local_z = int(z - self.hero_center_z + center)

        if 0 <= local_x < len(map_info) and 0 <= local_z < len(map_info[0]):
            return map_info[local_x][local_z] != 0  # 0表示障碍物
        return False

    def _compute_terrain_reward(self, terrain_feat):
        """根据地形特征计算奖励/惩罚"""
        rewards = []

        # 死角惩罚
        if terrain_feat[8] > 0:  # is_dead_end
            rewards.append(-0.2)

        # 走廊惩罚（走廊里容易被堵）
        if terrain_feat[9] > 0:  # is_corridor
            rewards.append(-0.05)

        # 开阔奖励（鼓励在开阔区域拉扯）
        if terrain_feat[10] > 0:  # is_open
            rewards.append(0.05)

        # 在死角时靠近墙根的惩罚
        min_depth = min(terrain_feat[:8])
        if min_depth < 0.2:  # 靠近墙
            rewards.append(-0.1 * (0.2 - min_depth))

        return sum(rewards)

    def _compute_real_distance(self, hero_pos, target_pos, map_info, entity_data):
        """
        复合真实距离：BFS优先，离散桶兜底

        BFS可达时用BFS真实路径距离，BFS不可达时（路径超出视野）用离散桶兜底
        归一化到 0-1 范围，21步以上视为1.0
        """
        if map_info is None:
            return 1.0

        if not entity_data.get("is_in_view", 0):
            return 1.0  # 看不到，返回最远

        center = len(map_info) // 2  # = 10
        hx = int(hero_pos["x"] - self.hero_center_x + center)
        hz = int(hero_pos["z"] - self.hero_center_z + center)
        tx = int(target_pos["x"] - self.hero_center_x + center)
        tz = int(target_pos["z"] - self.hero_center_z + center)

        # 边界检查
        if not (0 <= tx < len(map_info) and 0 <= tz < len(map_info[0])):
            return 1.0  # 超出视野，返回最远
        if map_info[tx][tz] == 0:
            return 1.0  # 目标位置不可通行，返回最远

        # BFS 计算真实路径距离
        bfs_dist = self._bfs((hx, hz), (tx, tz), map_info)

        if bfs_dist < float('inf'):
            return min(bfs_dist / 21.0, 1.0)  # 归一化，21步以上视为1.0
        else:
            # BFS不可达（路径超出视野），用离散桶兜底
            dist_bucket = entity_data.get("hero_l2_distance", 5) / 5.0
            return dist_bucket

    def _bfs(self, start, goal, map_info):
        """
        BFS 计算从起点到终点的最短路径距离（格子数）

        返回：路径长度，如果不可达返回 inf
        """
        from collections import deque

        sx, sz = start
        gx, gz = goal

        if map_info[sx][sz] == 0:
            return float('inf')  # 起点不可通行

        queue = deque([(sx, sz, 0)])
        visited = {(sx, sz)}
        directions = [(0, 1), (0, -1), (1, 0), (-1, 0)]

        while queue:
            x, z, dist = queue.popleft()
            if x == gx and z == gz:
                return dist

            for dx, dz in directions:
                nx, nz = x + dx, z + dz
                if (0 <= nx < len(map_info) and 0 <= nz < len(map_info[0])
                    and (nx, nz) not in visited
                    and map_info[nx][nz] != 0):
                    visited.add((nx, nz))
                    queue.append((nx, nz, dist + 1))

        return float('inf')  # 不可达

    def _process_legal_action(self, legal_act_raw):
        """处理16维合法动作掩码"""
        legal_action = [0] * 16
        if isinstance(legal_act_raw, list) and legal_act_raw:
            if isinstance(legal_act_raw[0], bool):
                for j in range(min(16, len(legal_act_raw))):
                    legal_action[j] = int(legal_act_raw[j])
            else:
                valid_set = {int(a) for a in legal_act_raw if int(a) < 16}
                legal_action = [1 if j in valid_set else 0 for j in range(16)]
        if sum(legal_action) == 0:
            legal_action = [1] * 16
        return np.array(legal_action, dtype=np.float32)

        # Critic head
